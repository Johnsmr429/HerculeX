cmake_minimum_required(VERSION 3.24)
project(surface LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

if(MLX_ROOT)
    unset(MLX_DIR CACHE)
    find_package(MLX CONFIG REQUIRED PATHS "${MLX_ROOT}" NO_DEFAULT_PATH)
else()
    find_package(MLX CONFIG QUIET)
    if(NOT MLX_FOUND)
        file(GLOB _mlx_roots "${CMAKE_CURRENT_SOURCE_DIR}/../../runtime/python/lib/python3*/site-packages/mlx")
        find_package(MLX CONFIG QUIET PATHS ${_mlx_roots} NO_DEFAULT_PATH)
    endif()
    if(NOT MLX_FOUND)
        message(FATAL_ERROR "Install the MLX C++ SDK and set HERCULEX_MLX_ROOT (or pass -DMLX_ROOT to CMake).")
    endif()
endif()
find_package(CURL REQUIRED)

add_executable(surface_predict src/main.cpp src/net.cpp src/predict.cpp src/volume.cpp src/fetch.cpp src/grid.cpp)
target_compile_options(surface_predict PRIVATE -O3 -mcpu=apple-m1 -ffp-contract=off)
target_include_directories(surface_predict PRIVATE ${MLX_INCLUDE_DIRS})
target_link_libraries(surface_predict PRIVATE mlx CURL::libcurl)
set_target_properties(surface_predict PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:mlx>")
